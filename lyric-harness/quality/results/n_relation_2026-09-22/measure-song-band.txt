"""Measure the `song` band profile from the committed, provenance-verified
N-relation rows (quality/results/n_relation_2026-09-22/calibration-rows.tsv).

The 2026-09-15 helper (quality/results/english_nonlyric_2026-09-15/measure-bands.txt)
with only the directory changed: the maintained band selector, author-held-out
checks and period checks, run over the rows' own five statistics; it
reconstructs only the item bodies and author dates the TSV omits, and refuses
unless every row matches its body by (tokens, file, title, lines, author).
Nothing is recomputed that the rows already hold.

    python3 quality/results/n_relation_2026-09-22/measure-song-band.txt song
"""
from pathlib import Path
import argparse,json,sys
ROOT=Path.cwd()
sys.path.insert(0,str(ROOT))
from quality import song_profile_calibration as C, length_curve_calibration as LC
p=argparse.ArgumentParser();p.add_argument('profile',choices=['song','short']);a=p.parse_args()
out=ROOT/'quality/results/n_relation_2026-09-22'
rowsfile=out/'calibration-rows.tsv'
provenance=LC.row_provenance();LC.verify_rows(rowsfile,provenance)
rows=LC.read_rows([str(rowsfile)],verify=True)
bodies=[]
for path in C.corpus_files():
 author,born,died=C.author_of(path)
 for title,body in C.items_in(path):
  if body:
   n_tokens=sum(len(C.QualityFeatures._tokens(line)) for line in body)
   if n_tokens:bodies.append((n_tokens,Path(path).name,title,body,author,born,died))
# read_rows sorts by length, filename and title. Match that stable order:
# repeated titles (e.g. Lovelace's "In English") are not unique work keys.
bodies.sort(key=lambda entry:entry[:3])
assert len(bodies)==len(rows)
for row,(n_tokens,filename,title,body,author,born,died) in zip(rows,bodies):
 assert (n_tokens,filename,title)==(row['n_tokens'],row['file'],row['title'])
 assert len(body)==row['n_lines'] and author==row['author']
 row.update(_body=body,born=born,died=died)
C.EDGES=list(C.PROFILE_EDGES[a.profile])
lo,hi=C.pick_band(rows,verbose=True,profile=a.profile)
full,fprs=C.report_fpr(rows,lo,hi,200)
slopes=C.report_period(rows,lo,hi,2000)
assert LC.row_provenance()==provenance
selected=C.band(rows,lo,hi)
record={'version':1,'profile':a.profile,'provenance':provenance,'lo':lo,'hi':hi,'items':len(selected),'files':len({r['file'] for r in selected}),'seeds':200,'draws':2000,'with_predictability':True,'sampled':None,'thresholds':full,'held_out_fpr':fprs,'period':slopes}
(out/(a.profile+'-adoption.json')).write_text(json.dumps(record,indent=2,sort_keys=True)+'\n')
print(json.dumps(record,indent=2))
